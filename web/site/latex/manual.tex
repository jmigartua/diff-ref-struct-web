% diff-ref-struct Web · the manuals' PDF preamble (tools/pdf.sh, LuaLaTeX through pandoc): the site's typefaces (the
% family kit's Source Sans 3 for the text, Source Serif 4 for the headings, IBM Plex Mono; all SIL OFL, fetched by
% pdf.sh into .fonts/), the app's green, symbols the faces lack taken from the system's fallbacks.
\usepackage{fontspec}
\directlua{
  luaotfload.add_fallback("drsfallback", {"Apple Symbols:mode=harf;", "Arial Unicode MS:mode=harf;", "Menlo:mode=harf;"})
}
\defaultfontfeatures{Ligatures=TeX}
\setmainfont{SourceSans3}[Path=\fontpath/, Extension=.otf, UprightFont=*-Regular, ItalicFont=*-It,
  BoldFont=*-Semibold, BoldItalicFont=*-SemiboldIt, RawFeature={fallback=drsfallback}]
\setsansfont{SourceSans3}[Path=\fontpath/, Extension=.otf, UprightFont=*-Regular, ItalicFont=*-It,
  BoldFont=*-Semibold, BoldItalicFont=*-SemiboldIt, RawFeature={fallback=drsfallback}]
\newfontfamily\drshead{SourceSerif4}[Path=\fontpath/, Extension=.otf, UprightFont=*-Semibold, ItalicFont=*-SemiboldIt,
  BoldFont=*-Bold, BoldItalicFont=*-BoldIt, RawFeature={fallback=drsfallback}]
\newfontfamily\drsserif{SourceSerif4}[Path=\fontpath/, Extension=.otf, UprightFont=*-Regular, ItalicFont=*-It,
  BoldFont=*-Semibold, RawFeature={fallback=drsfallback}]
\setmonofont{IBMPlexMono}[Path=\fontpath/, Extension=.otf, UprightFont=*-Regular, ItalicFont=*-Italic,
  BoldFont=*-Medium, Scale=0.88, RawFeature={fallback=drsfallback}]
